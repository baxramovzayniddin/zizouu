print("--- Welcome to the Smart Shopping Cart ---")
customer_name = input("Enter customer name: ")
is_member = input("Are you a loyalty member? (yes/no): ").strip().lower() == "yes"

print("\n--- Enter 3 Items ---")
# Item 1

item3_price = float(input("Item 3 price ($): "))

# --- 2. CALCULATE SUBTOTALS ---
total1 = item1_price * item1_qty
total2 = item2_price * item2_qty
